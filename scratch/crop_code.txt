const fs = require('fs');
const jpeg = require('jpeg-js');

const jpegData = fs.readFileSync('public/elden_ring_map.jpg');
const raw = jpeg.decode(jpegData, { useTArray: true, formatAsRGBA: true });
const w = raw.width;
const h = raw.height;
const data = raw.data;

// Crop region:
// x: 1750 to 2450 (width: 700)
// y: 3250 to 3850 (height: 600)
const minX = 1750, maxX = 2450;
const minY = 3250, maxY = 3850;
const cropW = maxX - minX;
const cropH = maxY - minY;
const cropBuf = Buffer.alloc(cropW * cropH * 4);

for (let y = minY; y < maxY; y++) {
  for (let x = minX; x < maxX; x++) {
    const srcIdx = (y * w + x) * 4;
    const destX = x - minX;
    const destY = y - minY;
    const destIdx = (destY * cropW + destX) * 4;
    
    // Copy RGB
    let r = data[srcIdx];
    let g = data[srcIdx+1];
    let b = data[srcIdx+2];
    
    // Grid lines every 100 pixels:
    // Leaflet lng = x, Leaflet lat = 5000 - y
    if (x % 100 === 0 || y % 100 === 0) {
      r = 255; g = 0; b = 0; // Red grid line
    }
    if (x % 50 === 0 && y % 50 === 0) {
      r = 255; g = 255; b = 0; // Yellow dot every 50
    }
    
    cropBuf[destIdx] = r;
    cropBuf[destIdx+1] = g;
    cropBuf[destIdx+2] = b;
    cropBuf[destIdx+3] = 255;
  }
}

const encoded = jpeg.encode({ data: cropBuf, width: cropW, height: cropH }, 85);
if (!fs.existsSync('scratch')) fs.mkdirSync('scratch');
fs.writeFileSync('scratch/limgrave_grid.jpg', encoded.data);
console.log('Saved scratch/limgrave_grid.jpg with coordinate grid!');
